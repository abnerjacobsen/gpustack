    ```python
        ProviderClientBase,
        CloudInstance,
        CloudInstanceCreate,
        InstanceState,
    )
    from gpustack.schemas.clusters import Volume

    logger = logging.getLogger(__name__)
    
    # AWS EC2 instance state mapping to InstanceState
    status_mapping = {
        "pending": InstanceState.CREATED,
        "running": InstanceState.RUNNING,
        "stopping": InstanceState.STOPPING,
        "stopped": InstanceState.STOPPED,
        "terminated": InstanceState.TERMINATED,
        "shutting-down": InstanceState.STOPPING,
    }
    
    class AWSClient(ProviderClientBase):
        """AWS EC2 cloud provider client using aiobotocore."""
        
        def __init__(self, access_key: str, secret_key: str, region: str, config: Optional[AWSConfig] = None):
            self.access_key = access_key
            self.secret_key = secret_key
            self.region = region
            self.config = config
            
            # Configure retry with max_attempts: 10 as per research
            self.boto_config = Config(
                retries={"max_attempts": 10, "mode": "adaptive"},
                connect_timeout=10,
                read_timeout=30,
            )
            self._session = get_session()
        
        async def _get_client(self):
            """Get aiobotocore EC2 client with proper configuration."""
            return self._session.create_client(
                "ec2",
                region_name=self.region,
                aws_access_key_id=self.access_key,
                aws_secret_access_key=self.secret_key,
                config=self.boto_config,
            )
        
        # Implement all ProviderClientBase abstract methods:
        # - create_instance
        # - delete_instance
        # - get_instance
        # - wait_for_started
        # - wait_for_public_ip
        # - create_ssh_key
        # - delete_ssh_key
        # - create_volumes_and_attach
        
        @classmethod
        def get_api_endpoint(cls) -> str:
            return f"https://ec2.{region}.amazonaws.com"
        
        @classmethod
        def process_header(cls, ak: str, sk: str, options: dict, headers: dict) -> dict:
            headers["Authorization"] = f"AWS4-HMAC-SHA256 Credential={ak}"
            return headers
    ```

    ```python
    def _handle_aws_error(self, error: Exception, operation: str) -> None:
        """Convert AWS exceptions to user-friendly error messages."""
        if isinstance(error, ClientError):
            error_code = error.response["Error"]["Code"]
            error_msg = error.response["Error"]["Message"]
            logger.error(f"AWS {operation} failed: {error_code} - {error_msg}")
            raise RuntimeError(f"AWS {operation} failed: {error_code}") from error
        elif isinstance(error, NoCredentialsError):
            logger.error(f"AWS credentials invalid for {operation}")
            raise RuntimeError("Invalid AWS credentials. Please check your access key and secret key.") from error
        else:
            logger.error(f"Unexpected error during {operation}: {error}")
            raise
    ```
