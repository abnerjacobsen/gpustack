    Add validate_credentials method to AWSClient for credential verification.
    
    Add method:
    ```python
    async def validate_credentials(self) -> bool:
        """
        Validate AWS credentials by calling STS GetCallerIdentity.
        Returns True if credentials are valid, raises RuntimeError otherwise.
        """
        try:
            async with self._get_client() as client:
                # Use STS GetCallerIdentity - lightweight auth check
                # Requires: sts:GetCallerIdentity permission
                response = await client.get_caller_identity()
                account = response.get("Account")
                arn = response.get("Arn")
                logger.info(f"AWS credentials validated for account {account}, ARN: {arn}")
                return True
        except ClientError as e:
            self._handle_aws_error(e, "credential validation")
        except NoCredentialsError as e:
            self._handle_aws_error(e, "credential validation")
        except Exception as e:
            logger.error(f"Unexpected error during credential validation: {e}")
            raise RuntimeError(f"Failed to validate AWS credentials: {str(e)}") from e
    ```
    
    Note: Since aiobotocore EC2 client doesn't have STS methods directly, 
    you may need to create an STS client OR use EC2 describe_regions as a 
    lightweight validation alternative:
    
    ```python
    async def validate_credentials(self) -> bool:
        """Validate by calling EC2 describe_regions (lightweight read operation)."""
        try:
            async with self._get_client() as client:
                response = await client.describe_regions(RegionNames=[self.region])
                if response.get("Regions"):
                    logger.info(f"AWS credentials validated for region {self.region}")
                    return True
                raise RuntimeError("AWS credentials validation failed: no regions returned")
        except (ClientError, NoCredentialsError) as e:
            self._handle_aws_error(e, "credential validation")
        ```
